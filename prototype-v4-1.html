CSS — all-or-nothing wrap using container queries threshold: … 
/* 1. Mark the wrapper as a query container */
.selects-container {
  container-type: inline-size;
}

/* 2. Default: selects side-by-side, equal width */
.selects-row {
  display: grid;
  grid-auto-flow: column;
  grid-auto-columns: 1fr;
  gap: 8px;
}
.selects-row .select {
  min-width: 160px;  /* per-select minimum */
}

/* 3. When container can't fit all of them, stack into one column */
/* threshold = count × min-width + gaps */
@container (max-width: 478px) {
  .selects-row {
    grid-auto-flow: row;
  }
  .selects-row .select {
    min-width: 0;
  }
}
